function hash(value: unknown) {
  return createHash('sha256').update(JSON.stringify(value)).digest('hex');
}

function difference(a: Buffer, b: Buffer) {
  const left = PNG.sync.read(a), right = PNG.sync.read(b);
  expect([right.width,right.height]).toEqual([left.width,left.height]);
  let count=0, minX=left.width, minY=left.height, maxX=-1, maxY=-1;
  const first: Array<{x:number;y:number;before:number[];after:number[]}> = [];
  for(let y=0;y<left.height;y++) for(let x=0;x<left.width;x++) {
    const at=4*(y*left.width+x);
    if(left.data[at]===right.data[at] && left.data[at+1]===right.data[at+1] &&
       left.data[at+2]===right.data[at+2] && left.data[at+3]===right.data[at+3]) continue;
    count++;minX=Math.min(minX,x);minY=Math.min(minY,y);maxX=Math.max(maxX,x);maxY=Math.max(maxY,y);
    if(first.length<30) first.push({x,y,before:Array.from(left.data.subarray(at,at+4)),after:Array.from(right.data.subarray(at,at+4))});
  }
  return {count,bounds:count?[minX,minY,maxX,maxY]:null,first};
}

    expect(raw.state).toEqual(initial.state);
    expect(raw.commands).toEqual(initial.commands);

      diagnosticsHash:hash(raw.diagnostics),reducedMotion:raw.reducedMotion,
