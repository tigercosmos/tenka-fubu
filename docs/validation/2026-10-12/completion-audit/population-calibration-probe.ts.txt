import { loadS1560Scenario } from '../../../../src/data/scenarios/s1560/index.ts';
import { BAL } from '../../../../src/core/balance.ts';
const bundle=await loadS1560Scenario();
const rows=bundle.districts.map(d=>({id:d.id,castleId:d.castleId,kokudaka:d.kokudaka,population:d.population,populationPerKoku:d.kokudaka>0?d.population/d.kokudaka:null,populationBasedSoldiers:Math.floor(d.population*BAL.soldiersPerPop),declaredKokuBasedSoldiers:d.kokudaka*0.025}));
